### Setup your Prisma schema

No `password` column — users exist by email alone.

```prisma lineNumbers title="/prisma/schema.prisma"
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "sqlite"
  url      = "file:./dev.db"
}

model User {
  id        String   @id @default(uuid())
  email     String   @unique
  name      String?
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}
```

<Step>
### Token utilities

Three token types: access (30m), refresh (30d), magic link (15m, single purpose).

```ts lineNumbers title="/features/auth/utils/tokens.ts"
import jwt from "jsonwebtoken";

export function generateAccessToken(userId: string): string {
  return jwt.sign({ id: userId }, process.env.JWT_SECRET!, {
    expiresIn: process.env.JWT_EXPIRES_IN,
  });
}

export function generateRefreshToken(userId: string): string {
  return jwt.sign({ id: userId }, process.env.JWT_REFRESH_SECRET!, {
    expiresIn: process.env.JWT_REFRESH_EXPIRES_IN,
  });
}

export function generateMagicLinkToken(userId: string): string {
  return jwt.sign({ id: userId }, process.env.MAGIC_LINK_SECRET!, {
    expiresIn: "15m",
  });
}

export function verifyToken(token: string, secret: string): unknown {
  return jwt.verify(token, secret);
}
```

</Step>

<Step>
### Email utility

```ts lineNumbers title="/features/auth/utils/email.ts"
import nodemailer from "nodemailer";

const transporter = nodemailer.createTransport({
  host: process.env.SMTP_HOST,
  port: Number(process.env.SMTP_PORT ?? 587),
  auth: {
    user: process.env.SMTP_USER,
    pass: process.env.SMTP_PASS,
  },
});

export async function sendMagicLink(email: string, token: string): Promise<void> {
  const frontendUrl = String(process.env.FRONTEND_URL ?? "http://localhost:3000");
  const link = `${frontendUrl}/verify?token=${token}`;

  await transporter.sendMail({
    from: process.env.SMTP_USER,
    to: email,
    subject: "Your magic link",
    html: `<p>Click <a href="${link}">here</a> to log in. This link expires in 15 minutes.</p>`,
  });
}
```

</Step>

<Step>
### Magic link flow

1. User enters email — `POST /auth/magic-link` upserts the user and emails a 15-minute JWT.
2. User clicks the link — the frontend POSTs the token to `/auth/verify-magic-link`.
3. Server verifies the magic-link JWT, issues the standard access/refresh cookie pair.

```ts lineNumbers title="/features/auth/controllers/magicLinkHandler.ts"
import { Request, Response } from "express";

import { prisma } from "@/config";
import { generateMagicLinkToken } from "@/features/auth/utils/tokens";
import { sendMagicLink } from "@/features/auth/utils/email";

export const magicLinkHandler = async (req: Request, res: Response) => {
  const { email } = req.body;

  const user = await prisma.user.upsert({
    where: { email },
    update: {},
    create: { email },
  });

  const token = generateMagicLinkToken(user.id);

  await sendMagicLink(email, token);

  return res.json({ message: "Magic link sent" }).status(200);
};
```

</Step>
</Steps>

The `verify-magic-link`, `refresh`, `me` and `logout` handlers mirror the [JWT Express example](/jwt/server/express) — once the magic link is exchanged, the session is a plain JWT cookie pair.
